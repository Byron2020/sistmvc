CREATE DATABASE inventario_db
CHARACTER SET utf8
COLLATE utf8_general_ci;

use inventario_db;

CREATE TABLE usuarios (
    id_usuario INT AUTO_INCREMENT PRIMARY KEY,
    cedu_usuario VARCHAR(50) NOT NULL UNIQUE,
    nomb_usuario VARCHAR(100) NOT NULL,
    pass_usuario VARCHAR(255) NOT NULL,
    tipo_usuario ENUM('Administrador','Gestion') NOT NULL DEFAULT 'Gestion',
    esta_usuario TINYINT(1) NOT NULL DEFAULT 1,
    fech_usuario TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);

CREATE TABLE t_productos (
  id_producto INT AUTO_INCREMENT PRIMARY KEY,
  nomb_producto VARCHAR(100) NOT NULL,
  desc_producto TEXT,
  precfact_producto DECIMAL(10,2) NOT NULL,
  precvent_producto DECIMAL(10,2) NOT NULL,
  stoc_producto INT NOT NULL DEFAULT 0,
  esta_producto TINYINT(1) DEFAULT 1
);

CREATE TABLE t_compras (
    id_compra INT AUTO_INCREMENT PRIMARY KEY,
    id_usuario INT,
    nume_factura VARCHAR(100) NULL,
    fech_factura DATE,
    subt_compra DECIMAL(10,2), 
    iva_compra DECIMAL(10,2),
    total_compra DECIMAL(10,2),
    esta_compra TINYINT DEFAULT 1,
    regi_compra TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);

CREATE TABLE t_cdetalle (
    id_cdetalle INT AUTO_INCREMENT PRIMARY KEY,
    id_compra INT,
    id_producto INT,
    cant_cdetalle INT,
    prec_cdetalle DECIMAL(10,2)
);

CREATE TABLE t_ventas (
    id_venta INT AUTO_INCREMENT PRIMARY KEY,
    id_usuario INT NOT NULL,
    clie_venta VARCHAR(100) NULL,
    fech_venta DATE NOT NULL,
    subt_venta DECIMAL(10,2) NOT NULL,
    iva_venta DECIMAL(10,2) NOT NULL,
    tota_venta DECIMAL(10,2) NOT NULL,
    esta_venta TINYINT DEFAULT 1,
    regi_venta TIMESTAMP DEFAULT CURRENT_TIMESTAMP,

    CONSTRAINT fk_venta_usuario
        FOREIGN KEY (id_usuario) REFERENCES t_usuarios(id_usuario)
);
CREATE TABLE t_vdetalle (
    id_vdetalle INT AUTO_INCREMENT PRIMARY KEY,
    id_venta INT NOT NULL,
    id_producto INT NOT NULL,
    cant_vdetalle INT NOT NULL,
    prec_vdetalle DECIMAL(10,2) NOT NULL,
    tota_vdetalle DECIMAL(10,2) NOT NULL,

    CONSTRAINT fk_vdetalle_venta
        FOREIGN KEY (id_venta) REFERENCES t_ventas(id_venta),

    CONSTRAINT fk_vdetalle_producto
        FOREIGN KEY (id_producto) REFERENCES t_productos(id_producto)
);
